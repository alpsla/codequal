```java
package org.apache.kafka.common.utils;

import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

public class Shell {
    // Whitelist allowed characters (adjust as needed)
    private static final Pattern SAFE_CHARS = Pattern.compile("^[a-zA-Z0-9._/-]+$");

    public static ProcessBuilder createProcessBuilder(String command, String... args) throws IOException {
        List<String> cmdList = new ArrayList<>();
        cmdList.add(command);

        for (String arg : args) {
            // Validate arguments to prevent injection
            if (!SAFE_CHARS.matcher(arg).matches())
                throw new IllegalArgumentException("Invalid characters in argument: " + arg);
            cmdList.add(arg);
        }

        return new ProcessBuilder(cmdList);
    }

    // Example usage with validation
    public static Process executeSafeCommand(String baseCmd, String userInputArg) throws IOException {
        // Validate user input before adding to command
        if (!SAFE_CHARS.matcher(userInputArg).matches())
            throw new IllegalArgumentException("Invalid user input: " + userInputArg);

        List<String> command = new ArrayList<>();
        command.add(baseCmd);
        command.add(userInputArg);

        ProcessBuilder pb = new ProcessBuilder(command);
        pb.redirectErrorStream(true); // Optional: combine error/output streams
        return pb.start();
    }
}
```

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.concurrent.ConcurrentHashMap;

public class Graph {
    private final ConcurrentHashMap<String, List<String>> adjacencyMap = new ConcurrentHashMap<>();

    /**
     * Adds directed edge from source to destination
     * @param source non-null node
     * @param destination non-null node
     * @throws IllegalArgumentException if either node is null
     */
    public void addEdge(String source, String destination) {
        if (source == null || destination == null) {
            throw new IllegalArgumentException("Nodes must be non-null");
        }

        // Atomic operation using ConcurrentHashMap's computeIfAbsent
        // Avoids race conditions and redundant null checks in concurrent environments
        adjacencyMap.computeIfAbsent(source, k -> new ArrayList<>()).add(destination);
    }

    /**
     * Returns unmodifiable list of neighbors for given node
     * @param node non-null target node
     * @return immutable List<String> or empty list if node not present
     */
    public List<String> getNeighbors(String node) {
        if (node == null) {
            throw new IllegalArgumentException("Node must be non-null");
        }
        // ConcurrentHashMap.get() is O(1) with low overhead
        List<String> neighbors = adjacencyMap.get(node);
        return neighbors == null ? Collections.emptyList() : Collections.unmodifiableList(neighbors);
    }
}
```
