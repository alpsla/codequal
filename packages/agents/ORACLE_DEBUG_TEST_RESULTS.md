```typescript
// In v9-integrated-analyzer.ts:129
public async analyzeRepository(
  repoUrl: string,
  prNumber: number,
  baseBranch: string = 'main'
): Promise<AnalysisResult> {
  // STRICT VALIDATION
  if (!prNumber || prNumber === 0 || !Number.isInteger(prNumber) || prNumber < 1) {
    const error = new Error(
      `[V9 VALIDATION ERROR] Invalid PR number: ${prNumber}\n` +
      `  Repository: ${repoUrl}\n` +
      `  Requirement: prNumber must be a positive integer > 0\n` +
      `  This error indicates the analyzer was called incorrectly.\n` +
      `  Stack trace follows...`
    );
    console.error(error.message);
    console.error(error.stack);
    throw error;
  }

  console.log(`[V9] Starting analysis: ${repoUrl} PR #${prNumber}`);
  // ... rest of function
}
```

**Why this works:**
- Catches invalid calls immediately
- Provides clear error message with context
- Shows stack trace to identify caller
- Prevents confusing reports from being generated
- Works in both local and Oracle environments
