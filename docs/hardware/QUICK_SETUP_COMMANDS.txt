# Quick Setup Commands for Oracle Linux 9 Instance

## Connect to Instance
ssh -i ~/Downloads/ssh-key-2025-05-08.key opc@129.213.49.128

## Once connected, run these commands:

# 1. Kill any stuck dnf processes (if needed)
sudo pkill dnf
sleep 2

# 2. Install Docker
sudo dnf install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin --nogpgcheck

# 3. Enable and start Docker
sudo systemctl enable docker
sudo systemctl start docker
sudo usermod -aG docker opc

# 4. Apply Docker group permissions
newgrp docker

# 5. Test Docker
docker run hello-world

# 6. Create workspace
sudo mkdir -p /mnt/workspace/{repos,output,cache}
sudo chown -R opc:opc /mnt/workspace

# 7. Create docker-compose.yml
cat > /mnt/workspace/docker-compose.yml << 'EOF'
version: "3.8"

services:
  analyzer-java:
    image: registry.digitalocean.com/codequal/analyzer:lang-java-v5.1
    cpuset: "0"
    mem_limit: 5g
    volumes:
      - /mnt/workspace/repos:/workspace/repos:ro
      - /mnt/workspace/output/java:/workspace/output
    command: ["/analyze.sh", "/workspace/repos/repo", "/workspace/output"]

  analyzer-python:
    image: registry.digitalocean.com/codequal/analyzer:lang-python-v4.3
    cpuset: "1"
    mem_limit: 5g
    volumes:
      - /mnt/workspace/repos:/workspace/repos:ro
      - /mnt/workspace/output/python:/workspace/output
    command: ["/analyze.sh", "/workspace/repos/repo", "/workspace/output"]

  analyzer-javascript:
    image: registry.digitalocean.com/codequal/analyzer:lang-javascript-v4.3
    cpuset: "2"
    mem_limit: 5g
    volumes:
      - /mnt/workspace/repos:/workspace/repos:ro
      - /mnt/workspace/output/javascript:/workspace/output
    command: ["/analyze.sh", "/workspace/repos/repo", "/workspace/output"]

  analyzer-security:
    image: registry.digitalocean.com/codequal/analyzer:security-v4.2
    cpuset: "3"
    mem_limit: 5g
    volumes:
      - /mnt/workspace/repos:/workspace/repos:ro
      - /mnt/workspace/output/security:/workspace/output
    command: ["/analyze.sh", "/workspace/repos/repo", "/workspace/output"]
EOF

# 8. Login to Docker Registry
docker login registry.digitalocean.com
# Enter your DigitalOcean email and API token

# 9. Pull images
docker pull registry.digitalocean.com/codequal/analyzer:lang-java-v5.1
docker pull registry.digitalocean.com/codequal/analyzer:lang-python-v4.3
docker pull registry.digitalocean.com/codequal/analyzer:lang-javascript-v4.3
docker pull registry.digitalocean.com/codequal/analyzer:security-v4.2

# 10. Create run script
cat > /mnt/workspace/run-analysis.sh << 'SCRIPT'
#!/bin/bash
REPO_URL=$1
BRANCH=${2:-main}

if [ -z "$REPO_URL" ]; then
    echo "Usage: $0 <repo-url> [branch]"
    exit 1
fi

echo "Analyzing: $REPO_URL ($BRANCH)"

# Clean & prepare
rm -rf /mnt/workspace/output/*
mkdir -p /mnt/workspace/output/{java,python,javascript,security}

# Clone repo
rm -rf /mnt/workspace/repos/repo
git clone --depth=10 --branch="$BRANCH" "$REPO_URL" /mnt/workspace/repos/repo

# Run analyzers
cd /mnt/workspace
docker compose up -d

echo "Analysis running... Monitor with: docker ps"
echo "View logs: docker logs <container-name>"
echo "Stop: docker compose down"
SCRIPT

chmod +x /mnt/workspace/run-analysis.sh

# 11. Test with Redis
/mnt/workspace/run-analysis.sh https://github.com/redis/redis unstable

# Monitor
docker ps
docker stats
